#### **环境管理**

```bash
# 创建环境
conda create -n env_name python=3.9
conda create -n env_name python=3.9 pandas numpy  # 指定包

# 激活/退出环境
conda activate env_name
conda deactivate

# 查看环境列表
conda env list
conda info --envs

# 删除环境
conda remove -n env_name --all

# 复制环境
conda create -n new_env --clone old_env

# 导出/导入环境
conda env export > environment.yaml  # 导出
conda env create -f environment.yaml  # 导入
```



#### 包管理

```bash
# 安装包
conda install package_name
conda install package_name=1.2.3  # 指定版本
conda install -c channel_name package_name  # 指定频道

# 卸载包
conda remove package_name

# 更新包
conda update package_name
conda update --all  # 更新所有包

# 查看已安装包
conda list
conda list -n env_name  # 查看指定环境的包

# 搜索包
conda search package_name
```



#### 配置管理

```bash
# 查看配置
conda info
conda config --show

# 添加/删除频道
conda config --add channels channel_name
conda config --remove channels channel_name

# 设置镜像源（国内常用）
conda config --add channels https://mirrors.tuna.tsinghua.edu.cn/anaconda/pkgs/main/
conda config --add channels https://mirrors.tuna.tsinghua.edu.cn/anaconda/pkgs/free/
conda config --set show_channel_urls yes

# 设置搜索时显示频道地址
conda config --set show_channel_urls yes
```



#### 清理与维护

```bash
# 清理缓存
conda clean --all
conda clean -p  # 清理未使用的包
conda clean -t  # 清理tar包

# 检查conda版本
conda --version

# 更新conda自身
conda update conda
conda update anaconda  # 更新Anaconda发行版
```



#### 常用组合命令

```bash
# 创建环境并指定Python版本
conda create -n py39 python=3.9 ipython jupyter

# 克隆base环境
conda create -n myenv --clone base

# 导出环境（跨平台）
conda env export --no-builds > environment.yml

# 查看环境详细信息
conda info -e  # 显示带*的当前环境
```
